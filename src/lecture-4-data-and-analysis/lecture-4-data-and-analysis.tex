	\subsection{Central Tendency}
	\begin{frame}
		\frametitle{Central Tendency}
		\begin{itemize}
			\item Arithmetic Mean:\\ $\bar{A} = \frac{1}{n} \sum_{i=1}^{n}a_i$\vspace{.125in}
			\item Geometric Mean\footnote{useful whenever the quantities to be averaged combine multiplicatively}:\\ $\bar{G} = \left(\prod_{i=1}^n x_i\right)^{\frac{1}{n}} = \sqrt[n]{x_1 x_2 \ldots x_n}$\vspace{.125in}
			\item Harmonic Mean\footnote{most appropriate average for ratios and rates such as speeds, robust to outliers}:\\ $\bar{H} == \frac{n}{\frac{1}{x_1} + \frac{1}{x_2} + \ldots \frac{1}{x_n}} = \frac{n}{\sum_{i=1}^n \frac{1}{x_i}}$\vspace{.125in}
			\item Median
			\item Mode
		\end{itemize}
	\end{frame}
	
	\subsection{Measures of Variance}
	\begin{frame}
		\frametitle{Measures of Variance}
		\begin{itemize}
			\item Range:\\ $\text{range} = max(X) - min(X)$\vspace{.125in}
			\item Variance:\\ $\text{Var}(X) = \sigma^2 = \frac{1}{n} \sum_{i=1}^n (x_i - \mu)^2$\vspace{.125in}
			\item Standard Deviation:\\ $\sigma = \sqrt{\sigma^2} = \sqrt{ \frac{1}{n} \sum_{i=1}^{n} (x_i - \mu)^2 }$\vspace{.125in}
			\item Pooled Standard Deviation:\\ $\sigma_p = \sqrt{ \frac{\sum_{i=1}^k (n_i - 1)\sigma_i^2}{\sum_{i=1}^k (n_i - 1)} } = \sqrt{ \frac{(n_1-1)\sigma_1^2 + (n_2-1)\sigma_2^2 + \ldots (n_k-1)\sigma_k^2}{n_1 + n_2 + \ldots + n_k - k} }$
		\end{itemize}
	\end{frame}

		Suppose that we have samples of blood pressure before a treatment, and then after.\\Likewise, what if we had two different groups we wanted to compare?

		We can use a t-test! This tests if the two samples are likely to have been sampled from the same population.

		\begin{columns}
			\begin{column}{.25\linewidth}

				$ t = \frac{(\mu_1 - mu_2)}{\sigma_p \cdot \sqrt{(\frac{1}{n_1}) + (\frac{1}{n_2})}} $ \vspace{.125in}

			\begin{column}{.6\linewidth}
				$N = \sum_{i=1}^k n_i = 15$

				%$\sum X = 182$

				%$\sum X^2 = 2404$

				Total Sum of Squares:

				$SS_T = \sum(X-\bar(X)^2)$

				$ = \sum X^2 - \frac{(\sum X)^2}{N} $ \vspace{.125in} % = 2404 - \frac{182^2}{15} = 2208.3$ \vspace{.125in}

				Sum of Squares Between Groups:

				$SS_B = \sum n_i (\bar{X_i} - \bar{X})^2 = \sum_{i=1}^k \left( \frac{(\sum X_i)^2}{n_i}\right) - \frac{(\sum X)^2}{N}$

				$ = \frac{48^2}{4}+\frac{83^2}{6}+\frac{51^2}{5} - \frac{182^2}{15} $ \vspace{.125in} %= 2244.4 - 2208.3 = 36.1$ \vspace{.125in}

				Sum of Squares Within Groups:\\ $SS_W = SS_T - SS_B$% = 2208.3 - 36.1 = 2172.2$
			\end{column}
		\end{columns}
	\end{frame}
