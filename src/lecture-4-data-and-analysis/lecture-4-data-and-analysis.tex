	\subsection{Central Tendency}
	\begin{frame}
		\frametitle{Central Tendency (commonly $\mu$)}
		\begin{itemize}
			\item Arithmetic Mean:\\ $\bar{A} = \frac{1}{n}\sum_{i=1}^{n}a_i$
			% \vspace{.25in}
			\item Geometric Mean\footnote{useful whenever the quantities to be averaged combine multiplicatively}:\\ $\bar{G} = \left(\prod_{i=1}^n x_i\right)^{\frac1n} = \sqrt[n]{x_1 x_2 \ldots x_n}$
			% \vspace{.25in}
			\item Harmonic Mean\footnote{most appropriate average for ratios and rates such as speeds, robust to outliers}:\\ $\bar{H} \
			frac{n}{\frac{1}{x_1} + \frac{1}{x_2} + \ldots \frac{1}{x_n}} = \frac{n}{\sum_{i=1}^n \frac{1}{x_i}}$
			\item Median
			\item Mode
		\end{itemize}
	\end{frame}
	
	\subsection{Measures of Variance}
	\begin{frame}
		\frametitle{Measures of Variance}
		\begin{itemize}
			\item Range:\\ $\text{Range} = max(X) - min(X)$
			% \vspace{.25in}
			\item Variance:\\ $\text{Var}(X) = \sigma^2 = \frac1n \sum_{i=1}^n (x_i - \mu)^2$
			% \vspace{.25in}
			\item Standard Deviation:\\ $\sigma = \sqrt{\sigma^2} = \sqrt{\frac1n \sum_{i=1}^n (x_i - \mu)^2$}$
			\item Pooled Standard Deviation:\\ $\sigma_p = \sqrt{\frac{\sum_{i=1}^k (n_i - 1)\sigma_i^2}{\sum_{i=1}^k (n_i - 1)}} = \sqrt{\frac{(n_1-1)\sigma_1^2 + (n_2-1)\sigma_2^2 + \ldots (n_k-1)\sigma_k^2}{n_1 + n_2 + \ldots + n_k - k}}$
		\end{itemize}
	\end{frame}
